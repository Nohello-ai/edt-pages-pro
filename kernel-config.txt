# ============================================================================
# 高通 Gunyah Hypervisor 虚拟化支持
# （DroidVM 基于 Gunyah 运行，这些选项是 Guest 与 Hypervisor 通信的基础）
# ============================================================================
CONFIG_GUNYAH=y                  # Gunyah 主驱动，检测并初始化 Hypervisor 接口
CONFIG_GUNYAH_PLATFORM=y         # 平台初始化，处理 Gunyah 设备树节点
CONFIG_GUNYAH_MESSAGING=y        # 消息队列，用于 Host 与 Guest 之间的数据传输

CONFIG_GUNYAH_DRIVERS=y          # Gunyah 外设驱动总开关

CONFIG_GUNYAH_RM=y               # Resource Manager，动态管理内存资源

# ============================================================================
# Gunyah 下的 Virtio 传输层

# ============================================================================
CONFIG_GUNYAH_VIRTIO_BE=y        # Gunyah 后端 virtio 传输接口（高通标准）
CONFIG_VIRTIO_MMIO=y             # 使能 MMIO 方式的 virtio 设备（Gunyah 默认）
CONFIG_VIRTIO_MMIO_CMDLINE_DEVICES=y  # 允许通过内核命令行指定 virtio 设备

# ============================================================================
# 必需的 Virtio 设备驱动

# ============================================================================
# 调度器与抢占（降低虚拟化延迟，提升响应速度）
# ============================================================================
